! ==========================================================
! SKENARIO 3: RIPv2 ROUTING CONFIGURATION
! Topologi : 3 Router (R-Tengah, R-Kiri, R-Kanan)
! ==========================================================


! ----------------------------------------------------------
! 1. ROUTER TENGAH
! ----------------------------------------------------------
enable
configure terminal
hostname R-Tengah

! Konfigurasi Interface
interface GigabitEthernet0/0
 description Link-to-R-Kiri
 ip address 200.10.10.1 255.255.255.252
 no shutdown
exit

interface GigabitEthernet0/1
 description Link-to-R-Kanan
 ip address 200.20.20.1 255.255.255.252
 no shutdown
exit

! Konfigurasi RIPv2
router rip
 version 2
 network 200.10.10.0
 network 200.20.20.0
 no auto-summary
exit

end
write memory


! ----------------------------------------------------------
! 2. ROUTER KIRI

exit

! Konfigurasi RIPv2
router rip
 version 2

exit

end
