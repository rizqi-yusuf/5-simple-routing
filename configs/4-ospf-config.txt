! ==========================================================
! SKENARIO 4: SINGLE-AREA OSPF CONFIGURATION (AREA 0)
! Topologi : 3 Router (R-Tengah, R-Kiri, R-Kanan)
! ==========================================================


! ----------------------------------------------------------
! 1. ROUTER TENGAH
! ----------------------------------------------------------
enable
configure terminal
hostname R-Tengah

! Konfigurasi Interface
interface GigabitEthernet0/0
 description Link-to-R-Kiri
 ip address 200.10.10.1 255.255.255.252
 no shutdown
exit

interface GigabitEthernet0/1
 description Link-to-R-Kanan
 ip address 200.20.20.1 255.255.255.252
 no shutdown
exit

! Konfigurasi OSPF Process ID 1 (Area 0)
router ospf 1
 network 200.10.10.0 0.0.0.3 area 0
 network 200.20.20.0 0.0.0.3 area 0
exit

end
write memory


! ----------------------------------------------------------
! 2. ROUTER KIRI
! ----------------------------------------------------------
enable
configure terminal
hostname R-Kiri

! Konfigurasi Interface
interface GigabitEthernet0/0
 description Link-to-R-Tengah
 ip address 200.10.10.2 255.255.255.252
 no shutdown
exit

interface GigabitEthernet0/1
 description Gateway-LAN-Kiri
 ip address 192.168.10.1 255.255.255.0
 no shutdown
exit

! Konfigurasi OSPF Process ID 1 (Area 0)
router ospf 1
 network 200.10.10.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
exit

end
write memory

exit

exit
